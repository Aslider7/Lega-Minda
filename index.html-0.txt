/* =========================================================
   WEB AUDIO API (CHIMES + AMBIENT SOUND STUDIO)
   ========================================================= */
let audioCtx = null;
const activeSounds = {};

function ensureAudio() {
  if (!audioCtx) {
    const AC = window.AudioContext || window.webkitAudioContext;
    audioCtx = new AC();
  }
  if (audioCtx.state === 'suspended') audioCtx.resume();
}

function playChime(freq = 523.25, dur = 0.25) {
  try {
    ensureAudio();
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = 'sine';
    osc.frequency.setValueAtTime(freq, audioCtx.currentTime);
    gain.gain.setValueAtTime(0.08, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.0001, audioCtx.currentTime + dur);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + dur);
  } catch (e) {}
}

function createNoiseNode(type) {
  const bufferSize = audioCtx.sampleRate * 2;
  const buffer = audioCtx.createBuffer(1, bufferSize, audioCtx.sampleRate);
  const data = buffer.getChannelData(0);
  let lastOut = 0;
  for (let i = 0; i < bufferSize; i++) {
    const white = Math.random() * 2 - 1;
    if (type === 'brown') {
      data[i] = (lastOut + 0.02 * white) / 1.02;
      lastOut = data[i];
      data[i] *= 3.2;
    } else {
      data[i] = white * 0.4;
    }
  }
  const src = audioCtx.createBufferSource();
  src.buffer = buffer;
  src.loop = true;

  const filter = audioCtx.createBiquadFilter();
  filter.type = 'lowpass';
  filter.frequency.value = type === 'brown' ? 360 : 1200;

  const gain = audioCtx.createGain();
  gain.gain.value = 0;

  src.connect(filter);
  filter.connect(gain);
  gain.connect(audioCtx.destination);
  src.start();
  return { src, gain };
}

function setSoundVol(type, val) {
  const v = Number(val) / 100;
  if ((type === 'drone' || type === 'chime') && !state.isPro && v > 0) {
    document.getElementById(type === 'drone' ? 'volDrone' : 'volChime').value = 0;
    openProModal();
    return;
  }
  ensureAudio();

  if (type === 'rain' || type === 'brown') {
    if (!activeSounds[type]) activeSounds[type] = createNoiseNode(type);
    activeSounds[type].gain.gain.setTargetAtTime(v * 0.28, audioCtx.currentTime, 0.1);
  } else if (type === 'drone') {
    if (!activeSounds.drone) {
      const o1 = audioCtx.createOscillator();
      const o2 = audioCtx.createOscillator();
      const g = audioCtx.createGain();
      o1.type = 'sine'; o1.frequency.value = 216;
      o2.type = 'sine'; o2.frequency.value = 432;
      g.gain.value = 0;
      o1.connect(g); o2.connect(g); g.connect(audioCtx.destination);
      o1.start(); o2.start();
      activeSounds.drone = { o1, o2, gain: g };
    }
    activeSounds.drone.gain.gain.setTargetAtTime(v * 0.08, audioCtx.currentTime, 0.15);
  } else if (type === 'chime') {
    if (activeSounds.chimeTimer) clearInterval(activeSounds.chimeTimer);
    if (v > 0) {
      const notes = [523.25, 587.33, 659.25, 783.99, 880.0];
      activeSounds.chimeTimer = setInterval(() => {
        const n = notes[Math.floor(Math.random() * notes.length)];
        playChime(n, 0.9);
      }, 2800);
    }
  }
}

function stopAllSounds() {
  ['volRain', 'volBrown', 'volDrone', 'volChime'].forEach(id => {
    const el = document.getElementById(id);
    if (el) el.value = 0;
  });

    if (k === 'chimeTimer') clearInterval(activeSounds[k]);
    else if (activeSounds[k].gain) activeSounds[k].gain.gain.setTargetAtTime(0, audioCtx.currentTime, 0.1);
  });
  showToast(state.lang === 'ms' ? 'Semua bunyi dihentikan' : 'All sounds stopped');
}

  { id: 'shredder', icon: '🌬️', ms: 'Bebas Serabut', en: 'Stress Releaser', check: s => s.shredsDone >= 1 },
  { id: 'goal3', icon: '🔥', ms: 'Konsisten Harian', en: 'Daily Rhythm', check: s => s.dailyDone >= 3 },
  { id: 'master', icon: '👑', ms: 'Minda Juara (200 XP)', en: 'Calm Champion (200 XP)', check: s => s.xp >= 200 }
];

function checkBadges() {
  BADGE_DEFS.forEach(b => {
    if (!state.badges.includes(b.id) && b.check(state)) {
      state.badges.push(b.id);
    }
  });
}

function renderBadges() {
  const grid = document.getElementById('badgeGrid');
  grid.innerHTML = BADGE_DEFS.map(b => {
    const ok = state.badges.includes(b.id);
    return `<div class="badge-item ${ok ? 'unlocked' : ''}">
      <div style="font-size:22px; margin-bottom:4px;">${b.icon}</div>
      <div style="font-size:12.5px; font-weight:700;">${state.lang === 'ms' ? b.ms : b.en}</div>
    </div>`;
  }).join('');
}

function renderMoodHistory() {
  const box = document.getElementById('moodHistoryList');
  if (!state.moodHistory.length) {
    box.innerHTML = `<p style="font-size:12.5px; color:var(--muted); margin:0;">${state.lang === 'ms' ? 'Belum ada rekod. Pilih emosi di halaman Hab.' : 'No logs yet. Tap your mood on the Hub screen.'}</p>`;
    return;
  }
  box.innerHTML = state.moodHistory.slice(0, 7).map(m => `
    <div class="history-item">
      <span>${m.emoji} <strong>${t('m_' + m.mood)}</strong></span>
      <span style="color:var(--muted); font-size:11.5px;">${m.time}</span>
    </div>
  `).join('');
}

let breathInterval = null;

function openGame(type) {
  const proOnly = ['eq', 'finance', 'memory', 'rpg'];
  if (proOnly.includes(type) && !state.isPro) {
    openProModal();
    return;
  }

  switchTab('game');
  if (type === 'breath') renderBreathGame();
  else if (type === 'english' || type === 'eq' || type === 'finance') renderQuizGame(type, 0);
  else if (type === 'memory') renderZenMemory();
  else if (type === 'rpg') renderRPG(0, { calm: 60, pro: 60, bound: 60 });
}

function toggleBreath() {
  const circle = document.getElementById('breathCircle');
  const btn = document.getElementById('breathBtn');
  const ms = state.lang === 'ms';
  if (breathInterval) {
    clearInterval(breathInterval);
    breathInterval = null;
    circle.className = 'breath-circle';
    circle.textContent = ms ? 'Selesai ✨' : 'Done ✨';
    btn.textContent = ms ? 'Mula Nafas Tenang' : 'Start Calming Breath';
    return;
  }

  let step = 0;
  const runStep = () => {
    if (step === 0) {
      circle.className = 'breath-circle inhale';
      circle.textContent = ms ? 'Tarik Nafas...' : 'Inhale...';
      playChime(396, 0.4);
      step = 1;
      breathInterval = setTimeout(runStep, 4000);
    } else if (step === 1) {
      circle.className = 'breath-circle hold';
      circle.textContent = ms ? 'Tahan Sebentar...' : 'Hold Gently...';
      step = 2;
      breathInterval = setTimeout(runStep, 4000);
    } else {
      circle.className = 'breath-circle exhale';
      circle.textContent = ms ? 'Hembus Perlahan...' : 'Exhale Slowly...';
      playChime(528, 0.5);
      addXP(15, true);
      step = 0;
      breathInterval = setTimeout(runStep, 6000);
    }
  };
  btn.textContent = ms ? 'Henti' : 'Stop';
  runStep();
}

function shredThought() {
  const ta = document.getElementById('shredText');
  if (!ta.value.trim()) return;
  ta.classList.add('dissolving');
  playChime(659.25, 0.6);
  setTimeout(() => {
    ta.value = '';
    ta.classList.remove('dissolving');
    state.shredsDone += 1;
    addXP(20, true);
    showToast(state.lang === 'ms' ? 'Fikiran dilepaskan (+20 XP) 🌿' : 'Thought released (+20 XP) 🌿');
  }, 800);
}

function answerQuiz(deckKey, idx, optIdx, btn) {
  const item = QUIZ_DATA[deckKey][idx];
  const buttons = document.querySelectorAll('#optList .option-btn');
  buttons.forEach(b => b.disabled = true);

  if (item.opts[optIdx].ok) {
    btn.classList.add('correct');
    addXP(20, true);
  } else {
    btn.classList.add('wrong');
    buttons.forEach((b, i) => { if (item.opts[i].ok) b.classList.add('correct'); });
    addXP(5, false);
  }
  document.getElementById('insightBox').style.display = 'block';
  document.getElementById('nextQuizBtn').style.display = 'block';
}

let zenSeq = [], zenUserIdx = 0, zenLocked = true;

function renderZenMemory() {
  const c = document.getElementById('gameContainer');
  const ms = state.lang === 'ms';
  c.innerHTML = `
    <h2 style="margin:0 0 6px; font-size:17px;">✨ ${t('g5_title')}</h2>
    <p id="zenStatus" style="margin:0 0 10px; font-size:12.5px; color:var(--muted);">
      ${ms ? 'Perhatikan cahaya yang menyala, kemudian tekan ikut urutan.' : 'Watch the soft glow, then tap the tiles in order.'}
    </p>
    <div class="zen-grid">
      ${Array.from({ length: 9 }, (_, i) => `<div class="zen-tile" id="zt-${i}" onclick="tapZenTile(${i})"></div>`).join('')}
    </div>
    <button class="btn btn-primary" onclick="startZenRound(3)">${ms ? 'Mula Pusingan Santai' : 'Start Calm Round'}</button>
  `;
}

function startZenRound(len) {
  zenSeq = Array.from({ length: len }, () => Math.floor(Math.random() * 9));
  zenUserIdx = 0;
  zenLocked = true;
  const notes = [392, 440, 493.88, 523.25, 587.33, 659.25, 783.99, 880, 987.77];

  zenSeq.forEach((tileIdx, i) => {
    setTimeout(() => {
      const el = document.getElementById(`zt-${tileIdx}`);
      if (!el) return;
      el.classList.add('lit');
      playChime(notes[tileIdx], 0.35);
      setTimeout(() => el.classList.remove('lit'), 480);
      if (i === zenSeq.length - 1) {
        setTimeout(() => {
          zenLocked = false;
          const st = document.getElementById('zenStatus');
          if (st) st.textContent = state.lang === 'ms' ? 'Giliran anda! Tekan dengan tenang.' : 'Your turn! Tap calmly.';
        }, 550);
      }
    }, i * 750 + 300);
  });
}

function tapZenTile(i) {
  if (zenLocked) return;
  const el = document.getElementById(`zt-${i}`);
  el.classList.add('lit');
  setTimeout(() => el.classList.remove('lit'), 250);
  playChime(523.25 + i * 35, 0.25);

  if (i === zenSeq[zenUserIdx]) {
    zenUserIdx++;
    if (zenUserIdx === zenSeq.length) {
      zenLocked = true;
      addXP(25, true);
      showToast(state.lang === 'ms' ? 'Fokus mantap! (+25 XP) ✨' : 'Great focus! (+25 XP) ✨');
      setTimeout(() => startZenRound(Math.min(zenSeq.length + 1, 6)), 1000);
    }
  } else {
    zenLocked = true;
    showToast(state.lang === 'ms' ? 'Tak apa, cuba lagi dengan santai 🌿' : 'No worries, try again gently 🌿');
  }
}
